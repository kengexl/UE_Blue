```gdscript
class_name ConnectionData

@export var source_port_id: int
@export var target_component_id: int
@export var target_port_id: int
@export var connection_type: String = "exec"

func is_valid() -> bool:
    # 验证连接合法性:
    # 1. source_port 必须是 OUTPUT
    # 2. target_port 必须是 INPUT
    # 3. 不能自连接

        target_component_id, target_port_id
```

### 2.4 GraphData (图数据 - 顶层容器)

**文件**: `addons/blueprint_system/data/GraphData.gd`

```gdscript
class_name GraphData
extends Resource

@export var graph_name: String = "New Graph"
@export var graph_id: int
@export var components: Dictionary[int, ComponentData] = {}
@export var connections: Array[ConnectionData] = []
@export var metadata: Dictionary = {}
@export var editor_state: Dictionary = {}  # 视口偏移、缩放等

func add_component(data: ComponentData) -> void:
    components[data.component_id] = data

func remove_component(component_id: int) -> void:
    components.erase(component_id)
    # 同时移除该组件相关的所有连接

func add_connection(data: ConnectionData) -> bool:
    if not data.is_valid():
        return false
    connections.append(data)
    return true

func remove_connection(connection_id: int) -> void:
    connections = connections.filter(
        func(c): return c.connection_id != connection_id
    )
```

### 3.6 ComponentSystem (组件系统控制器)

**文件**: `addons/blueprint_system/core/ComponentSystem.gd`

```gdscript
class_name ComponentSystem
extends Node

signal component_registered(component_id: int)
signal component_unregistered(component_id: int)

var _components: Dictionary[int, BlueprintComponent] = {}
var _next_component_id: int = 1

func register_component(component: BlueprintComponent) -> bool:
    component.component_id = _next_component_id++
    _components[component.component_id] = component
    component_registered.emit(component.component_id)
    return true

func unregister_component(component_id: int) -> void:
    var comp = _components.get(component_id)
    if comp:
        comp.queue_free()
        _components.erase(component_id)
        component_unregistered.emit(component_id)

func get_component(component_id: int) -> BlueprintComponent:
    return _components.get(component_id)

func get_all_components() -> Array[BlueprintComponent]:
    return _components.values()

func create_component(type: int, position: Vector2, name: String = "") -> BlueprintComponent:
    var comp: BlueprintComponent
    match type:
        ComponentData.ComponentType.OUTPUT_ONLY:
            comp = TypeAComponent.new()
        ComponentData.ComponentType.DUAL:
            comp = TestComponent.new()
        ComponentData.ComponentType.INPUT_ONLY:
            comp = TypeCComponent.new()

    if not name.is_empty():
        comp.component_name = name
    comp.position = position
    register_component(comp)
    return comp
```

## 五、交互系统实现

### 5.1 InteractionSystem (交互系统控制器)

**文件**: `addons/blueprint_system/core/InteractionSystem.gd`

```gdscript
class_name InteractionSystem
extends Node

enum InteractionMode { NONE, DRAGGING_COMPONENT, CONNECTING_PORTS }

var current_mode: InteractionMode = InteractionMode.NONE
var _drag_data: Dictionary = {}

# ── 拖拽组件 ──
var _drag_component: BlueprintComponent
var _drag_offset: Vector2

# ── 连接操作 ──
var _source_port: PortWidget
var _source_component: BlueprintComponent
var _temp_line: Line2D  # 拖拽过程中的临时连线

# ── 输入事件处理 ──
func handle_canvas_input(event: InputEvent) -> void:
    if event is InputEventMouseButton:
        if event.button_index == MOUSE_BUTTON_RIGHT and event.pressed:
            # 右键点击→弹出上下文菜单（未点击组件时）
            if not _is_clicking_on_component(event.position):
                BlueprintEventBus.context_menu_requested.emit(
                    event.global_position, event.position
                )
        elif event.button_index == MOUSE_BUTTON_LEFT:
            if not event.pressed:
                # 释放左键
                _end_current_operation(event.position)

func handle_component_input(component: BlueprintComponent, event: InputEvent) -> void:
    if event is InputEventMouseButton:
        if event.button_index == MOUSE_BUTTON_LEFT and event.pressed:
            # 开始拖拽组件
            _start_drag_component(component, event.position)

func handle_port_input(port: PortWidget, event: InputEvent) -> void:
    if event is InputEventMouseButton:
        if event.button_index == MOUSE_BUTTON_LEFT and event.pressed:
            if current_mode == InteractionMode.NONE:
                # 开始连接操作
                _start_port_connection(port)
            elif current_mode == InteractionMode.CONNECTING_PORTS:
                # 尝试完成连接
                _try_complete_connection(port)

func _start_port_connection(port: PortWidget) -> void:
    if port.port_definition.direction == PortDefinition.Direction.INPUT:
        return  # 只能从输出端口开始拖拽
    current_mode = InteractionMode.CONNECTING_PORTS
    _source_port = port
    _source_component = port.parent_component
    port.set_highlight(true)
    # 创建临时连线跟随鼠标
    _temp_line = Line2D.new()
    # ...

func _try_complete_connection(target_port: PortWidget) -> void:
    if target_port == _source_port:
        _cancel_connection()
        return
    if target_port.port_definition.direction == PortDefinition.Direction.OUTPUT:
        _cancel_connection()
        return  # 不能连接到输出端口

    # 委托 ConnectionManager 验证并创建连接
    var result = ConnectionManager.validate_and_connect(
        _source_component.component_id, _source_port.port_definition.port_id,
        target_port.parent_component.component_id, target_port.port_definition.port_id
    )

    if result.success:
        BlueprintEventBus.connection_created.emit(result.connection)
    else:
        # 显示连接失败提示
        push_warning("Connection failed: " + result.error_message)

    _cancel_connection()

func _cancel_connection() -> void:
    if _source_port:
        _source_port.set_highlight(false)
    if _temp_line:
        _temp_line.queue_free()
    current_mode = InteractionMode.NONE
    _source_port = null
    _source_component = null
    _temp_line = null
```

### 7.2 ConnectionManager (连接管理器)

**文件**: `addons/blueprint_system/core/ConnectionManager.gd`

```gdscript
class_name ConnectionManager
extends Node

signal connection_created(connection: ConnectionData, line: ConnectionLine)
signal connection_removed(connection_id: int)

var _connections: Dictionary[int, ConnectionData] = {}
var _connection_lines: Dictionary[int, ConnectionLine] = {}
var _component_connections: Dictionary[int, Array[int]] = {}  # comp_id → [conn_ids]

# ── 验证并创建连接 ──
func validate_and_connect(
    src_comp_id: int, src_port_id: int,
    tgt_comp_id: int, tgt_port_id: int
) -> Dictionary:
    # 返回: { success: bool, error_message: String, connection: ConnectionData }

    # 1. 不能自连接
    if src_comp_id == tgt_comp_id:
        return { "success": false, "error_message": "Cannot self-connect" }

    # 2. 检查重复连接
    var hash = "%d_%d_%d_%d" % [src_comp_id, src_port_id, tgt_comp_id, tgt_port_id]
    for conn in _connections.values():
        if conn.get_hash() == hash:
            return { "success": false, "error_message": "Duplicate connection" }

    # 3. 创建连接
    var conn = ConnectionData.new()
    conn.connection_id = _connections.size() + 1
    conn.source_component_id = src_comp_id
    conn.source_port_id = src_port_id
    conn.target_component_id = tgt_comp_id
    conn.target_port_id = tgt_port_id
    _connections[conn.connection_id] = conn

    # 4. 创建连接线
    var line = ConnectionLine.new()
    line.connection_data = conn
    # 获取端口引用
    var src_comp = ComponentSystem.get_component(src_comp_id)
    var tgt_comp = ComponentSystem.get_component(tgt_comp_id)
    line.source_port = src_comp.get_port_widget_by_id(src_port_id)
    line.target_port = tgt_comp.get_port_widget_by_id(tgt_port_id)
    line.update_position()

    _connection_lines[conn.connection_id] = line
    _component_connections[src_comp_id] = _component_connections.get(
        src_comp_id, []
    ) + [conn.connection_id]
    _component_connections[tgt_comp_id] = _component_connections.get(
        tgt_comp_id, []
    ) + [conn.connection_id]

    connection_created.emit(conn, line)
    return { "success": true, "error_message": "", "connection": conn }

func remove_connection(connection_id: int) -> void:
    var conn = _connections.get(connection_id)
    if not conn:
        return

    # 从连接映射中移除
    _connections.erase(connection_id)
    for comp_id in [conn.source_component_id, conn.target_component_id]:
        if _component_connections.has(comp_id):
            _component_connections[comp_id].erase(connection_id)

    # 移除连接线
    if _connection_lines.has(connection_id):
        _connection_lines[connection_id].queue_free()
        _connection_lines.erase(connection_id)

    connection_removed.emit(connection_id)
```
